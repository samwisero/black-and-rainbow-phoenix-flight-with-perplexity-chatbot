        function randInt(a, b) { 
            return Math.floor(Math.random() * (b - a) + a); 
        }

        class Player {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 30;
                this.height = 60;
                this.velX = 0;
                this.velY = 0;
                this.onGround = false;
                this.facingRight = true;
                this.health = 15;
                this.maxHealth = 15;
                this.shootCooldown = 0;
                this.onFire = false;
                this.fireTimer = 0;
                this.fireSpeed = FIRE_RUN_SPEED;
                this.exploding = false;
                this.explosionTimer = 0;
                this.pieces = [];
                this.gameOver = false;
                this.arrows = [];
            }

            update(platforms) {
                if (this.exploding) {
                    this.explosionTimer++;
                    this.pieces.forEach(p => {
                        p.x += p.vx;
                        p.y += p.vy;
                        p.vy += GRAVITY * 0.15;
                    });
                    this.pieces = this.pieces.filter(p => 
                        p.x > -100 && p.x < SCREEN_WIDTH + 100 && 
                        p.y > -100 && p.y < SCREEN_HEIGHT + 100
                    );
                    if (this.pieces.length === 0) {
                        this.exploding = false;
                        this.gameOver = true;
                    }
                    return;
                }

                if (this.onFire) {
                    this.fireTimer++;
                    this.velX = this.fireSpeed * (this.facingRight ? 1 : -1);
                    if (this.fireTimer > 150) { // 2.5 seconds at 60fps
                        this.onFire = false;
                        this.fireTimer = 0;
                        this.velX = 0;
                        this.triggerExplosion();
                        return;
                    }
                }

                // Physics
                if (!this.onGround) {
                    this.velY += GRAVITY;
                }

                this.x += this.velX;
                this.y += this.velY;

                // Platform collisions
                this.onGround = false;
                for (const platform of platforms) {
                    if (this.velY > 0 && 
                        this.x + this.width > platform.x && 
                        this.x < platform.x + platform.width &&
                        this.y + this.height > platform.y && 
                        this.y + this.height < platform.y + platform.height + 10) {
                        this.y = platform.y - this.height;
                        this.velY = 0;
                        this.onGround = true;
                    }
                }

                // Ground collision
                if (this.y + this.height >= SCREEN_HEIGHT - 50) {
                    this.y = SCREEN_HEIGHT - 50 - this.height;
                    this.velY = 0;
                    this.onGround = true;
                }

                // Screen boundaries
                if (this.x < 0) this.x = 0;

                // Update cooldowns
                if (this.shootCooldown > 0) this.shootCooldown--;

                // Update health bar
                const healthBar = document.getElementById('gameHealthFill');
                if (healthBar) {
                    const healthPercent = Math.max(0, (this.health / this.maxHealth) * 100);
                    healthBar.style.width = healthPercent + '%';
                }
            }

            jump() {
                if (this.onGround && !this.onFire) {
                    this.velY = JUMP_STRENGTH;
                }
            }

            shoot() {
                if (this.shootCooldown === 0 && !this.onFire && !this.exploding) {
                    const arrow = new Arrow(
                        this.x + (this.facingRight ? this.width : 0),
                        this.y + 20,
                        this.facingRight
                    );
                    this.arrows.push(arrow);
                    this.shootCooldown = 20;
                }
            }

            triggerExplosion() {
                const baseX = this.x + this.width / 2;
                const baseY = this.y + this.height / 2;
                const colors = ["#ff0000", "#ff8000", "#ffff00", "#ff00ff"];
                
                for (let i = 0; i < 8; i++) {
                    const angle = Math.random() * 2 * Math.PI;
                    const speed = randInt(8, 15);
                    this.pieces.push({
                        x: baseX,
                        y: baseY,
                        vx: speed * Math.cos(angle),
                        vy: speed * Math.sin(angle) - randInt(2, 7),
                        radius: randInt(5, 12),
                        color: colors[randInt(0, colors.length)]
                    });
                }
                this.exploding = true;
                this.explosionTimer = 0;
            }

            draw(ctx) {
                const headX = this.x + this.width / 2;
                const headY = this.y + 15;

                if (this.exploding) {
                    for (const piece of this.pieces) {
                        ctx.beginPath();
                        ctx.arc(piece.x, piece.y, piece.radius, 0, Math.PI * 2);
                        ctx.fillStyle = piece.color;
                        ctx.fill();
                    }
                    return;
                }

                // Fire effect
                if (this.onFire) {
                    const fireIntensity = Math.min(this.fireTimer / 30, 1);
                    for (let i = 0; i < 5; i++) {
                        ctx.save();
                        ctx.globalAlpha = (1 - i * 0.2) * fireIntensity;
                        ctx.beginPath();
                        ctx.arc(headX + randInt(-15, 15), headY + randInt(-10, 20), randInt(8, 15), 0, Math.PI * 2);
                        ctx.fillStyle = i % 2 === 0 ? '#ff6400' : '#fff000';
                        ctx.fill();
                        ctx.restore();
                    }
                }

                ctx.save();
                ctx.strokeStyle = '#000000';
                ctx.lineWidth = 2;

                // Head
                ctx.beginPath();
                ctx.arc(headX, headY, 10, 0, Math.PI * 2);
                ctx.stroke();

                // Body
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(headX, headY + 10);
                ctx.lineTo(headX, this.y + 40);
                ctx.stroke();

                // Arms
                const armY = headY + 15;
                if (this.facingRight) {
                    ctx.beginPath();
                    ctx.moveTo(headX, armY);
                    ctx.lineTo(headX + 15, armY + 5);
                    ctx.stroke();
                    
                    // Bow
                    ctx.strokeStyle = '#8B4513';
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    ctx.arc(headX + 15, armY + 5, 10, -1.5, 1.5, false);
                    ctx.stroke();
                } else {
                    ctx.beginPath();
                    ctx.moveTo(headX, armY);
                    ctx.lineTo(headX - 15, armY + 5);
                    ctx.stroke();
                    
                    // Bow
                    ctx.strokeStyle = '#8B4513';
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    ctx.arc(headX - 15, armY + 5, 10, 1.6, 4.7, false);
                    ctx.stroke();
                }

                // Other arm
                ctx.strokeStyle = '#000000';
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(headX, armY);
                ctx.lineTo(headX + (this.facingRight ? -10 : 10), armY + 10);
                ctx.stroke();

                // Legs
                const legY = this.y + 40;
                ctx.beginPath();
                ctx.moveTo(headX, legY);
                ctx.lineTo(headX - 10, this.y + this.height);
                ctx.stroke();

                ctx.beginPath();
                ctx.moveTo(headX, legY);
                ctx.lineTo(headX + 10, this.y + this.height);
                ctx.stroke();

                ctx.restore();
            }
        }

        class Arrow {
            constructor(x, y, facingRight) {
                this.x = x;
                this.y = y;
                this.velX = facingRight ? ARROW_SPEED : -ARROW_SPEED;
                this.active = true;
                this.stuck = false;
                this.stuckTarget = null;
                this.offsetX = 0;
                this.offsetY = 0;
            }

            update() {
                if (!this.stuck) {
                    this.x += this.velX;
                    if (this.x < -50 || this.x > SCREEN_WIDTH + 50) {
                        this.active = false;
                    }
                } else if (this.stuckTarget) {
                    this.x = this.stuckTarget.x + this.offsetX;
                    this.y = this.stuckTarget.y + this.offsetY;
                }
            }

            draw(ctx) {
                ctx.save();
                ctx.strokeStyle = '#8B4513';
                ctx.lineWidth = 2;
                ctx.beginPath();
                const endX = this.x + (this.velX > 0 ? 15 : -15);
                ctx.moveTo(this.x, this.y);
                ctx.lineTo(endX, this.y);
                ctx.stroke();

                // Arrowhead
                ctx.fillStyle = '#000000';
                if (this.velX > 0) {
                    ctx.beginPath();
                    ctx.moveTo(endX, this.y);
                    ctx.lineTo(endX - 5, this.y - 3);
                    ctx.lineTo(endX - 5, this.y + 3);
                    ctx.fill();
                } else {
                    ctx.beginPath();
                    ctx.moveTo(endX, this.y);
                    ctx.lineTo(endX + 5, this.y - 3);
                    ctx.lineTo(endX + 5, this.y + 3);
                    ctx.fill();
                }
                ctx.restore();
            }
        }

        class Piggy {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 60;
                this.height = 50;
                this.velX = Math.random() > 0.5 ? PIG_SPEED : -PIG_SPEED;
                this.velY = 0;
                this.onGround = false;
                this.alive = true;
                this.angryTimer = randInt(0, 60);
            }

            update(platforms) {
                if (!this.alive) {
                    this.velY += GRAVITY * 0.7;
                    this.x += this.velX * 0.8;
                    this.y += this.velY;
                    return;
                }

                this.velY += GRAVITY * 0.7;
                this.x += this.velX;
                this.y += this.velY;

                // Random direction change
                if (randInt(0, 120) < 2) {
                    this.velX *= -1;
                    this.angryTimer = 60;
                }

                // Platform collisions
                this.onGround = false;
                for (const platform of platforms) {
                    if (this.velY > 0 && 
                        this.x + this.width > platform.x && 
                        this.x < platform.x + platform.width &&
                        this.y + this.height > platform.y && 
                        this.y + this.height < platform.y + platform.height + 10) {
                        this.y = platform.y - this.height;
                        this.velY = 0;
                        this.onGround = true;
                    }
                }

                // Ground collision
                if (this.y + this.height >= SCREEN_HEIGHT - 50) {
                    this.y = SCREEN_HEIGHT - 50 - this.height;
                    this.velY = 0;
                    this.onGround = true;
                }

                // Screen boundaries
                if (this.x <= 0 || this.x + this.width >= SCREEN_WIDTH) {
                    this.velX *= -1;
                    this.x = Math.max(0, Math.min(SCREEN_WIDTH - this.width, this.x));
                }

                if (this.angryTimer > 0) this.angryTimer--;
            }

            draw(ctx) {
                const centerX = this.x + this.width / 2;
                const centerY = this.y + this.height / 2;
                const color = this.alive ? (this.angryTimer > 0 ? '#FF4444' : '#FFC0CB') : '#666666';

                ctx.save();
                ctx.fillStyle = color;
                ctx.beginPath();
                ctx.ellipse(centerX, centerY, this.width / 2, this.height / 2, 0, 0, Math.PI * 2);
                ctx.fill();

                if (this.alive) {
                    // Snout
                    ctx.fillStyle = '#FFB6C1';
                    ctx.beginPath();
                    ctx.ellipse(centerX, centerY + 5, 12, 8, 0, 0, Math.PI * 2);
                    ctx.fill();

                    // Nostrils
                    ctx.fillStyle = '#000000';
                    ctx.beginPath();
                    ctx.arc(centerX - 3, centerY + 5, 1.5, 0, Math.PI * 2);
                    ctx.arc(centerX + 3, centerY + 5, 1.5, 0, Math.PI * 2);
                    ctx.fill();

                    // Eyes
                    ctx.fillStyle = '#FFFFFF';
                    ctx.beginPath();
                    ctx.arc(centerX - 12, centerY - 8, 6, 0, Math.PI * 2);
                    ctx.arc(centerX + 12, centerY - 8, 6, 0, Math.PI * 2);
                    ctx.fill();

                    ctx.fillStyle = '#000000';
                    ctx.beginPath();
                    ctx.arc(centerX - 12, centerY - 8, 3, 0, Math.PI * 2);
                    ctx.arc(centerX + 12, centerY - 8, 3, 0, Math.PI * 2);
                    ctx.fill();

                    // Angry eyebrows
                    if (this.angryTimer > 0) {
                        ctx.strokeStyle = '#000000';
                        ctx.lineWidth = 2;
                        ctx.beginPath();
                        ctx.moveTo(centerX - 18, centerY - 15);
                        ctx.lineTo(centerX - 8, centerY - 12);
                        ctx.moveTo(centerX + 8, centerY - 12);
                        ctx.lineTo(centerX + 18, centerY - 15);
                        ctx.stroke();
                    }
                }
                ctx.restore();
            }
        }

        class CloudPlatform {
            constructor(x, y, width) {
                this.x = x;
                this.y = y;
                this.width = width;
                this.height = 20;
            }

            draw(ctx) {
                ctx.save();
                ctx.fillStyle = '#FFFFFF';
                
                // Draw fluffy cloud shapes
                const cloudY = this.y + 5;
                for (let i = 0; i < this.width; i += 25) {
                    const cloudX = this.x + i;
                    ctx.beginPath();
                    ctx.arc(cloudX, cloudY, 15, 0, Math.PI * 2);
                    ctx.fill();
                    
                    ctx.beginPath();
                    ctx.arc(cloudX + 10, cloudY - 5, 12, 0, Math.PI * 2);
                    ctx.fill();
                    
                    ctx.beginPath();
                    ctx.arc(cloudX + 20, cloudY, 15, 0, Math.PI * 2);
                    ctx.fill();
                }

                // Base platform (invisible collision)
                ctx.globalAlpha = 0.1;
                ctx.fillStyle = '#CCCCCC';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                ctx.restore();
            }
        }

        // Game initialization
        function initGame() {
            gameState.canvas = document.getElementById('gameCanvas');
            gameState.ctx = gameState.canvas.getContext('2d');
            gameState.worldOffset = 0;
            
            // Create player
            gameState.player = new Player(100, 350);
            
            // Create fewer enemies (reduced from 9 to 5)
            gameState.enemies = [];
            for (let i = 0; i < MAX_PIGGIES; i++) {
                gameState.enemies.push(new Piggy(randInt(350, 800), randInt(200, 400)));
            }
            
            // Create platforms (actual cloud platforms)
            gameState.platforms = [
                new CloudPlatform(200, 450, 150),
                new CloudPlatform(550, 320, 120),
                new CloudPlatform(400, 200, 180),
                new CloudPlatform(1200, 300, 100)
            ];

            gameState.gameRunning = true;
            gameLoop();
        }

        function drawBonfire(ctx, x, y) {
            ctx.save();
            
            // Wood base
            ctx.fillStyle = '#8B4513';
            ctx.fillRect(x - 25, y, 50, 8);
            
            // Wood logs
            for (let i = 0; i < 3; i++) {
                ctx.fillRect(x - 20 + i * 20, y - 8, 8, 12);
            }
            
            // Flames
            for (let i = 0; i < 7; i++) {
                const flameX = x - 20 + i * 6;
                const flameHeight = 35 + Math.sin(Date.now() * 0.01 + i) * 5;
                
                ctx.beginPath();
                ctx.moveTo(flameX - 5, y);
                ctx.lineTo(flameX, y - flameHeight);
                ctx.lineTo(flameX + 5, y);
                ctx.closePath();
                
                const colors = ['#FF6400', '#FF8C00', '#FFD700'];
                ctx.fillStyle = colors[i % colors.length];
                ctx.fill();
            }
            ctx.restore();
        }

        function drawBackground(ctx) {
            // Sky gradient
            const gradient = ctx.createLinearGradient(0, 0, 0, SCREEN_HEIGHT);
            gradient.addColorStop(0, '#87CEEB');
            gradient.addColorStop(1, '#E0F6FF');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT);

            // Sun
            ctx.fillStyle = '#FFFF00';
            ctx.beginPath();
            ctx.arc(900, 80, 40, 0, Math.PI * 2);
            ctx.fill();

            // Ground
            ctx.fillStyle = '#228B22';
            ctx.fillRect(0, SCREEN_HEIGHT - 50, SCREEN_WIDTH, 50);

            // Trees (added back)
            for (let i = 0; i < 8; i++) {
                const treeX = i * 130 + 80 - gameState.worldOffset;
                if (treeX > -50 && treeX < SCREEN_WIDTH + 50) {
                    // Trunk
                    ctx.fillStyle = '#8B4513';
                    ctx.fillRect(treeX, SCREEN_HEIGHT - 90, 15, 40);
                    
                    // Leaves
                    ctx.fillStyle = '#228B22';
                    ctx.beginPath();
                    ctx.arc(treeX + 7, SCREEN_HEIGHT - 90, 25, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }

        function gameLoop() {
            if (!gameState.gameRunning) return;

            const ctx = gameState.ctx;
            
            // Clear canvas
            ctx.clearRect(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT);
            
            // Draw background
            drawBackground(ctx);
            
            // Draw platforms
            gameState.platforms.forEach(platform => platform.draw(ctx));
            
            // Draw bonfire
            drawBonfire(ctx, 1200 - gameState.worldOffset, SCREEN_HEIGHT - 50);
            
            // Update and draw player
            if (!gameState.player.gameOver) {
                gameState.player.update(gameState.platforms);
            }
            gameState.player.draw(ctx);
            
            // Update and draw arrows
            gameState.player.arrows = gameState.player.arrows.filter(arrow => {
                arrow.update();
                if (arrow.active) {
                    arrow.draw(ctx);
                    
                    // Check arrow-enemy collisions
                    gameState.enemies.forEach(enemy => {
                        if (enemy.alive && !arrow.stuck &&
                            arrow.x > enemy.x && arrow.x < enemy.x + enemy.width &&
                            arrow.y > enemy.y && arrow.y < enemy.y + enemy.height) {
                            enemy.alive = false;
                            arrow.stuck = true;
                            arrow.stuckTarget = enemy;
                            arrow.offsetX = arrow.x - enemy.x;
                            arrow.offsetY = arrow.y - enemy.y;
                            arrow.velX = 0;
                            
                            // Make pig fly when hit
                            enemy.velY = -PIG_BOUNCE_H;
                            enemy.velX = randInt(10, 15) * (arrow.velX > 0 ? 1 : -1);
                        }
                    });
                    
                    return true;
                }
                return false;
            });
            
            // Update and draw enemies
            gameState.enemies.forEach(enemy => {
                enemy.update(gameState.platforms);
                enemy.draw(ctx);
                
                // Check player-enemy collisions
                if (enemy.alive && !gameState.player.exploding && !gameState.player.onFire &&
                    gameState.player.x < enemy.x + enemy.width &&
                    gameState.player.x + gameState.player.width > enemy.x &&
                    gameState.player.y < enemy.y + enemy.height &&
                    gameState.player.y + gameState.player.height > enemy.y) {
                    
                    // Bounce player away
                    if (gameState.player.x < enemy.x) {
                        gameState.player.velX = -PIG_BOUNCE_H;
                    } else {
                        gameState.player.velX = PIG_BOUNCE_H;
                    }
                    gameState.player.velY = -PIG_BOUNCE_V;
                    enemy.angryTimer = 90;
                    gameState.player.health -= 2;
                }
            });
            
            // Check bonfire collision
            const bonfireX = 1200 - gameState.worldOffset;
            const bonfireW = 50;
            const bonfireTop = SCREEN_HEIGHT - 50;
            
            if (!gameState.player.onFire && !gameState.player.exploding &&
                gameState.player.x + gameState.player.width > bonfireX - bonfireW/2 &&
                gameState.player.x < bonfireX + bonfireW/2 &&
                gameState.player.y + gameState.player.height >= bonfireTop) {
                gameState.player.onFire = true;
                gameState.player.fireTimer = 0;
                if (gameState.player.velX === 0) {
                    gameState.player.facingRight = true;
                }
            }
            
            // Check game over
            if (gameState.player.health <= 0 && !gameState.player.gameOver) {
                gameState.player.gameOver = true;
            }
            
            // Game over screen
            if (gameState.player.gameOver && !gameState.player.exploding) {
                ctx.save();
                ctx.font = 'bold 48px Arial';
                ctx.fillStyle = '#FF0000';
                ctx.textAlign = 'center';
                ctx.fillText('GAME OVER', SCREEN_WIDTH / 2, SCREEN_HEIGHT / 2);
                ctx.font = '20px Arial';
                ctx.fillStyle = '#FFFFFF';
                ctx.fillText('Restarting in 3 seconds...', SCREEN_WIDTH / 2, SCREEN_HEIGHT / 2 + 40);
                ctx.restore();
                
                setTimeout(() => {
                    initGame();
                }, 3000);
                return;
            }
            
            // Side scrolling
            const centerX = SCREEN_WIDTH * 0.6;
            if (gameState.player.x > centerX && gameState.player.velX > 0) {
                const scroll = gameState.player.velX;
                gameState.worldOffset += scroll;
                gameState.platforms.forEach(p => p.x -= scroll);
                gameState.enemies.forEach(e => e.x -= scroll);
                gameState.player.x = centerX;
            }
            
            requestAnimationFrame(gameLoop);
        }

        function saveSystemPrompt() {
            const name = document.getElementById('promptName').value.trim();
            const content = document.getElementById('promptContent').value.trim();
            
            if (!name || !content) {
                alert('Please enter both a name and content for the system prompt.');
                return;
            }
            
            const newPrompt = {
                id: Date.now().toString(),
                name: name,
                content: content,
                createdAt: new Date()
            };
            
            appState.systemPrompts.push(newPrompt);
            localStorage.setItem('phoenixSystemPrompts', JSON.stringify(appState.systemPrompts));
            
            // Clear form
            document.getElementById('promptName').value = '';
            document.getElementById('promptContent').value = '';
            
            updateSavedPromptsList();
            alert('System prompt saved successfully!');
        }

        function updateSavedPromptsList() {
            const select = document.getElementById('savedPrompts');
            select.innerHTML = '<option value="">Select a prompt...</option>';
            
            appState.systemPrompts.forEach(prompt => {
                const option = document.createElement('option');
                option.value = prompt.id;
                option.textContent = prompt.name;
                select.appendChild(option);
            });
        }

        function loadSystemPrompt() {
            const selectedId = document.getElementById('savedPrompts').value;
            if (!selectedId) {
                alert('Please select a prompt to load.');
                return;
            }
            
            const prompt = appState.systemPrompts.find(p => p.id === selectedId);
            if (prompt) {
                appState.activeSystemPrompt = prompt.content;
                appState.currentSystemPromptId = prompt.id;
                localStorage.setItem('activeSystemPrompt', prompt.content);
                localStorage.setItem('currentSystemPromptId', prompt.id);
                updateCurrentPromptDisplay();
                alert(`System prompt "${prompt.name}" is now active!`);
            }
        }

        function deleteSystemPrompt() {
            const selectedId = document.getElementById('savedPrompts').value;
            if (!selectedId) {
                alert('Please select a prompt to delete.');
                return;
            }
            
            const prompt = appState.systemPrompts.find(p => p.id === selectedId);
            if (prompt && confirm(`Are you sure you want to delete "${prompt.name}"?`)) {
                appState.systemPrompts = appState.systemPrompts.filter(p => p.id !== selectedId);
                localStorage.setItem('phoenixSystemPrompts', JSON.stringify(appState.systemPrompts));
                
                // If this was the active prompt, clear it
                if (appState.currentSystemPromptId === selectedId) {
                    appState.activeSystemPrompt = null;
                    appState.currentSystemPromptId = null;
                    localStorage.removeItem('activeSystemPrompt');
                    localStorage.removeItem('currentSystemPromptId');
                    updateCurrentPromptDisplay();
                }
                
                updateSavedPromptsList();
                alert('System prompt deleted successfully!');
            }
        }

        function updateCurrentPromptDisplay() {
            const display = document.getElementById('currentPromptDisplay');
            if (appState.activeSystemPrompt) {
                const prompt = appState.systemPrompts.find(p => p.id === appState.currentSystemPromptId);
                const promptName = prompt ? prompt.name : 'Unknown';
                display.innerHTML = `
                    <div style="color: #4caf50;">Active: ${promptName}</div>
                    <div style="margin-top: 5px; font-size: 10px; opacity: 0.8;">${appState.activeSystemPrompt.substring(0, 100)}${appState.activeSystemPrompt.length > 100 ? '...' : ''}</div>
                `;
            } else {
                display.innerHTML = '<div>No system prompt active</div>';
            }
        }

        // REAL API CALL FUNCTION WITH SYSTEM PROMPT SUPPORT
        async function callRealAPI(message) {
            const provider = appState.selectedProvider;
            const model = appState.selectedModel;
            const apiKey = appState.apiKeys[provider];

            if (!apiKey) {
                return 'Error: No API key configured for ' + provider + '. Please set up your API keys in the admin panel.';
            }

            try {
                let apiUrl, headers, body;
                const messages = [];
                
                // Add system prompt if active
                if (appState.activeSystemPrompt) {
                    messages.push({
                        role: "system",
                        content: appState.activeSystemPrompt
                    });
                }
                
                // Add user message
                messages.push({
                    role: "user",
                    content: message
                });

                if (provider === 'cerebras') {
                    apiUrl = 'https://api.cerebras.ai/v1/chat/completions';
                    headers = {
                        'Authorization': `Bearer ${apiKey}`,
                        'Content-Type': 'application/json'
                    };
                    body = JSON.stringify({
                        model: model,
                        messages: messages,
                        max_tokens: 500,
                        temperature: 0.7
                    });
                } else if (provider === 'groq') {
                    apiUrl = 'https://api.groq.com/openai/v1/chat/completions';
                    headers = {
                        'Authorization': `Bearer ${apiKey}`,
                        'Content-Type': 'application/json'
                    };
                    body = JSON.stringify({
                        model: model,
                        messages: messages,
                        max_tokens: 500,
                        temperature: 0.7
                    });
                } else if (provider === 'openrouter') {
                    apiUrl = 'https://openrouter.ai/api/v1/chat/completions';
                    headers = {
                        'Authorization': `Bearer ${apiKey}`,
                        'Content-Type': 'application/json',
                        'HTTP-Referer': window.location.href,
                        'X-Title': 'Phoenix Flight Chat'
                    };
                    body = JSON.stringify({
                        model: model,
                        messages: messages,
                        max_tokens: 500,
                        temperature: 0.7
                    });
                } else {
                    return 'Error: Provider ' + provider + ' API calls not implemented yet.';
                }

                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: headers,
                    body: body
                });

                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }

                const data = await response.json();
                return data.choices[0].message.content;

            } catch (error) {
                console.error('API Error:', error);
                return `Error calling ${provider} API: ${error.message}. Please check your API key and try again.`;
            }
        }

        async function sendMessage() {
            const input = document.getElementById('messageInput');
            const message = input.value.trim();
            
            if (!message) return;
            
            // Add user message
            addMessage('user', message);
            input.value = '';
            
            // Show typing indicator
            showTypingIndicator();
            
            try {
                // Call real API instead of generic response
                const aiResponse = await callRealAPI(message);
                
                hideTypingIndicator();
                addMessage('assistant', aiResponse);
                
                // Simulate TTS
                simulateTTS();
            } catch (error) {
                hideTypingIndicator();
                addMessage('assistant', 'I apologize, but I encountered an error processing your request. Please check your API configuration and try again.');
            }
        }

                e.preventDefault(); // Prevent page scroll

        // Game Controls
        document.addEventListener('keydown', function(e) {
            if (gameState.gameRunning && appState.currentPage === 'game') {
                gameState.keys[e.key] = true;
                
                // Handle game controls
                if (gameState.player && !gameState.player.exploding && !gameState.player.onFire && !gameState.player.gameOver) {
                    gameState.player.velX = 0;
                    
                    if (e.key === 'ArrowLeft') {
                        gameState.player.velX = -MOVE_SPEED;
                        gameState.player.facingRight = false;
                    }
                    if (e.key === 'ArrowRight') {
                        gameState.player.velX = MOVE_SPEED;
                        gameState.player.facingRight = true;
                    }
                    if (e.key === 'ArrowUp') {
                        gameState.player.jump();
                    }
                    if (e.key === ' ') {
                        gameState.player.shoot();
                        e.preventDefault(); // Prevent scroll
                    }
                }
            }
        });

            if (gameState.gameRunning && appState.currentPage === 'game') {
                gameState.keys[e.key] = false;
            }
